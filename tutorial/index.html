"""Engine-neutral display v1 projection. No database, HTTP or model dependency."""
import math
from copy import deepcopy

SCHEMA_VERSION = "1.0"
ANGLE_CONVENTION = "long_axis_clockwise_from_positive_x_mod_180"
ENGINE_LABELS = {"pytorch": "PyTorch · YOLOv5-OBB", "openvino": "OpenVINO", "tensorrt": "TensorRT"}


def finite(value, *, nullable=True):
    if value is None and nullable:
        return None
    if isinstance(value, bool) or not isinstance(value, (int, float)) or not math.isfinite(value):
        raise ValueError("协议数值必须有限")
    return value


def detection_view(item, geometry_type="obb"):
    points = item.get("polygon_px", item.get("polygon"))
    if not isinstance(points, (list, tuple)) or len(points) != 4:
        raise ValueError("二维检测必须提供四个顺序角点")
    polygon = []
    for point in points:
        if not isinstance(point, (list, tuple)) or len(point) != 2:
            raise ValueError("角点必须为 [x,y]")
        polygon.append([float(finite(v, nullable=False)) for v in point])
    confidence = float(finite(item.get("confidence"), nullable=False))
    if not 0 <= confidence <= 1:
        raise ValueError("置信度必须在 0..1")
    edges = [(polygon[(i+1) % 4][0]-polygon[i][0], polygon[(i+1) % 4][1]-polygon[i][1]) for i in range(4)]
    lengths = [math.hypot(*edge) for edge in edges]
    if min(lengths) <= 0:
        raise ValueError("检测框包含退化边")
    crosses = [edges[i][0]*edges[(i+1) % 4][1]-edges[i][1]*edges[(i+1) % 4][0] for i in range(4)]
    if not (all(c > 0 for c in crosses) or all(c < 0 for c in crosses)):
        raise ValueError("四角点必须依序组成非退化凸四边形")
    # Only numerical square degeneracy is rejected; task-specific near-square
    # reliability is not calibrated, so this is a geometric direction, not a pose.
    longest = max(range(4), key=lambda i: lengths[i])
    valid = geometry_type == "obb" and not math.isclose(max(lengths), min(lengths), rel_tol=1e-6, abs_tol=1e-8)
    angle = math.degrees(math.atan2(edges[longest][1], edges[longest][0])) % 180 if valid else None
    return {"id": item["id"], "class_id": item["class_id"], "class_name": str(item["class_name"]),
            "confidence": confidence, "geometry_type": geometry_type, "polygon_px": polygon,
            "center_px": [sum(p[a] for p in polygon)/4 for a in (0, 1)],
            "angle_deg": angle, "angle_convention": ANGLE_CONVENTION,
            "orientation_valid": valid, "track_id": None}


def safe_error(value, code):
    # Internal exception text may contain filesystem paths. It stays in admin
    # records/logs; the read-only public display gets a stable diagnostic code.
    if not value:
        return None
    return {"code": code, "message": "处理失败，请在后台查看诊断；当前状态不代表未检出目标。"}


def stream_view(run):
    if run is None:
        return None
    model = run.get("model") or {}
    engine = (run.get("config") or {}).get("engine", "")
    classes = [{k: c.get(k) for k in ("id", "name", "label", "color")} for c in model.get("classes", [])]
    return {"session_id": run["id"], "source_id": run["source_id"], "source_label": run["source_label"],
            "mode": "continuous" if run.get("total") is None else "finite", "status": run["status"],
            "model": {"id": model.get("name"), "version": model.get("sha256"),
                      "name": model.get("name"), "device": model.get("device")},
            "engine_label": ENGINE_LABELS.get(engine, str(engine)), "classes": classes,
            "class_catalog_version": model.get("sha256"),
            **{k: deepcopy(run.get(k)) for k in ("class_counts", "processed", "total", "succeeded", "failed", "timings")},
            "error": safe_error(run.get("error"), "RUN_FAILED")}


def frame_view(run, frame):
    if frame is None or frame.get("run_id") != run["id"]:
        return None
    key = frame.get("capture_key") or {"source_id": run["source_id"], "session_id": run["id"], "frame_id": str(frame["id"])}
    key = {k: str(key[k]) for k in ("source_id", "session_id", "frame_id")}
    if not all(key.values()):
        raise ValueError("帧身份字段不能为空")
    geometry_type = (run.get("model") or {}).get("box_type", "obb")
    return {"frame_key": key, "stream_session_id": run["id"], "id": str(frame["id"]),
            "name": frame["name"], "status": frame["status"], "width": frame["width"], "height": frame["height"],
            "coordinate_space": "source_pixels", "detections": [detection_view(d, geometry_type) for d in frame["detections"]],
            "timing_ms": deepcopy(frame.get("timing_ms", {})), "created_at": frame.get("created_at"),
            "capture_time": frame.get("capture_time"), "capture_clock_domain": frame.get("capture_clock_domain"),
            "published_at": frame.get("published_at"), "preview": None,
            "preview_status": "pending" if frame["status"] == "ok" else "none",
            "error": safe_error(frame.get("error"), "FRAME_FAILED")}

打开可操作的画布实验

关键代码的含义

polygon.map(p=>p.join(',')).join(' ') 将 [[x,y],...] 转为 x,y x,y ...。viewBox 定义内部坐标范围，CSS width 定义屏幕显示宽度。两者分开，所以缩放页面时不需要重算每个点。

vector-effect="non-scaling-stroke" 让框线宽度不随缩放成倍变粗。这改变绘制效果，不改变检测坐标。真实组件随后会用 <image> 显示原图；当前实验只证明几何关系。

逐句理解

@app.post 把 URL 和函数绑定；函数参数的类型告诉 FastAPI 如何解析 JSON。Field(ge=...,le=...) 在进入函数前拒绝非法参数。model_dump() 将请求模型转回普通字典，方便保存和返回。

    def __init__(self, root, receipt_path):
        self.root = Path(root).resolve()
        if not self.root.is_dir():
            raise InputError("相机接收目录不存在，请先运行模拟拍照脚本")
        receipt_path = Path(receipt_path)
        receipt_path.parent.mkdir(parents=True, exist_ok=True)
        self.receipts = sqlite3.connect(receipt_path)
        self.receipts.execute("CREATE TABLE IF NOT EXISTS receipts (name TEXT PRIMARY KEY, digest TEXT NOT NULL)")
        self.receipts.execute("CREATE TABLE IF NOT EXISTS identities (identity TEXT PRIMARY KEY, name TEXT NOT NULL)")
        self.pending = []
        self.pending_count = 0
        self.last_scan = 0.0

    def _read(self, path):
        if path.is_symlink() or path.stat().st_size > 16384:
            raise InputError("相机发布清单无效：符号链接或清单超过16KiB")
        payload = path.read_bytes()
        digest = hashlib.sha256(payload).hexdigest()
        done = self.receipts.execute("SELECT digest FROM receipts WHERE name=?", (path.name,)).fetchone()
        if done:
            if done[0] != digest:
                raise InputError("已消费清单被修改；请使用新的帧身份和文件名")
            return None
        try:
            meta = json.loads(payload)
            if meta.get("schema_version") != "1.0":
                raise ValueError("schema")
            name = meta["image"]
            if not isinstance(name, str) or name != Path(name).name or '/' in name or '\\' in name:
                raise ValueError("image name")
            image = self.root / name
            if image.is_symlink() or image.resolve().parent != self.root or image.suffix.lower() not in EXTENSIONS:
                raise ValueError("image path")
            key = {k: str(meta[k]) for k in ("source_id", "session_id", "frame_id")}
            if not all(re.fullmatch(r"[A-Za-z0-9_-]{1,100}", v) for v in key.values()):
                raise ValueError("identity")
            if not key["frame_id"].isdigit():
                raise ValueError("sequence")
            identity = json.dumps(key, sort_keys=True)
            if self.receipts.execute("SELECT 1 FROM identities WHERE identity=?", (identity,)).fetchone():
                raise ValueError("previously consumed identity")
            expected = meta["sha256"]
            if not isinstance(expected, str) or not re.fullmatch(r"[a-f0-9]{64}", expected):
                raise ValueError("digest")
            published = datetime.fromisoformat(meta["published_at"])
            if published.tzinfo is None:
                raise ValueError("timezone")
            captured = meta.get("capture_time")
            if captured is not None and datetime.fromisoformat(captured).tzinfo is None:
                raise ValueError("capture timezone")
            stat = image.stat()
            if type(meta["size"]) is not int or stat.st_size != meta["size"] or not 0 < stat.st_size <= LIMITS["max_file_bytes"]:
                raise ValueError("size")
        except (KeyError, ValueError, TypeError, OSError, AttributeError) as exc:
            raise InputError("相机发布清单或对应图片无效；请检查身份、时间、完整写入及20MiB上限") from exc
        return {"path": image, "name": name, "size": stat.st_size, "mtime": stat.st_mtime_ns,

                "expected_sha256": expected, "manifest_name": path.name, "manifest_digest": digest,
                "_order": (published.timestamp(), key["session_id"], int(key["frame_id"]))}

    def next_item(self):
        if not self.pending:
            now = time.monotonic()
            if now - self.last_scan < .1:
                return None
            self.last_scan = now
            found = []
            identities = set()

            for path in self.root.glob("*.ready.json"):

                if item is None:
                    continue
                identity = tuple(item["capture_key"].values())
                if identity in identities:
                    raise InputError("不同清单重复使用同一个相机帧身份")
                identities.add(identity)
                found.append(item)
                if len(found) > LIMITS["max_images"]:
                    raise InputError("相机未消费清单超过10000条，停止任务；未删除或跳过源图片")
            self.pending = sorted(found, key=lambda item: item["_order"], reverse=True)
        self.pending_count = len(self.pending)
        return self.pending.pop() if self.pending else None

    def ack(self, item):

        self.receipts.execute("INSERT INTO receipts(name,digest) VALUES(?,?)",

        self.receipts.commit()
        self.pending_count = len(self.pending)

        self.receipts.close()

    def _publish(self, frame=None, image_path=None):

    def source(self, source_id):

            src = self.settings.sources[source_id]

    def control(self, run_id, action):
        with self.condition:
            run = self.current
            if not run or run["id"] != run_id:
                raise Conflict("只能控制当前活动任务")
            status = run["status"]
            if action == "stop" and status in {"stopping", "stopped"}:
                return copy.deepcopy(run)
            if status in TERMINAL:
                raise Conflict("任务已结束；请创建新任务重新运行")
            if action == "pause":
                if status not in {"running", "pausing", "paused"}:
                    raise Conflict("仅运行中的任务可以暂停")
                if status == "running":
                    run["status"] = "pausing"
            elif action == "resume":
                if status not in {"running", "paused", "pausing"}:
                    raise Conflict("当前状态不可继续")
                run["status"] = "running"
            else:
                run["status"] = "stopping"
            # Accept an in-memory intent. The single worker persists transitions
            # at a frame boundary; a slow database cannot hold the state lock.
            self._control_dirty = True
            self._publish()
            self.condition.notify_all()
            return copy.deepcopy(run)

            return {"run": copy.deepcopy(self.current), "frame": copy.deepcopy(self.latest_frame)}

    def _boundary(self):
        while True:
            transition = None
            with self.condition:
                if self.current["status"] == "pausing":
                    self.current["status"] = "paused"
                    self._publish()
                if self.current["status"] == "stopping":
                    return False
                if self._control_dirty:
                    transition = copy.deepcopy(self.current)
                    self._control_dirty = False
                if transition is None and self.current["status"] != "paused":
                    return True
            if transition is not None:
                self.store.save_run(transition)
            with self.condition:
                if self.current["status"] == "paused" and not self._control_dirty:
                    self.condition.wait()

    def _work(self, files):
        run = self.current
        continuous = isinstance(files, dict) and "inbox_root" in files
        times = deque(maxlen=1000) if continuous else []
        total_time = success_count = 0
        try:
            if continuous:

                files = InboxSource(files["inbox_root"], files["receipt_path"])

                else:

                model = adapter.load()
                self.adapter, self.model = adapter, model
            with self.condition:
                run["model"] = copy.deepcopy(self.model)
                if run["status"] != "stopping":

                initial = copy.deepcopy(run)
                self._publish()
            self.store.save_run(initial)
            output = self.settings.runtime / "runs" / run["id"] / "images"
            output.mkdir(parents=True, exist_ok=True)
            index = 0
            while True:
                if not self._boundary():
                    break
                if continuous:
                    item = files.next_item()
                    if item is None:
                        with self.condition:
                            if run["status"] == "running":
                                self.condition.wait(timeout=.1)
                        continue
                else:
                    if index >= len(files):
                        break
                    item = files[index]

                    updated = copy.deepcopy(run)
                    updated["processed"] += 1
                    updated["latest_frame_id"] = index

                    if frame["status"] == "ok":
                        times.append(frame["timing_ms"]["total"])
                        total_time += frame["timing_ms"]["total"]
                        success_count += 1
                        mean = total_time / success_count
                        updated["timings"] = {"mean_ms": mean, "p50_ms": float(np.percentile(times, 50)),

            except Exception:

                    self.subscribers.pop(token, None)

    def _deliver(self, token):
        with self.condition:
            sub = self.subscribers.get(token)
            if not sub:
                return
            sub["scheduled"] = False
            snapshot = self._snapshot()
            queue = sub["queue"]
            if queue.full():
                queue.get_nowait()
            queue.put_nowait(snapshot)

    def subscribe(self, loop):
        with self.condition:
            if len(self.subscribers) >= self.MAX_SUBSCRIBERS or self.closed:
                raise RuntimeError("展示订阅已达上限或服务正在关闭")
            token, queue = uuid.uuid4().hex, asyncio.Queue(maxsize=1)
            self.subscribers[token] = {"loop": loop, "queue": queue, "scheduled": False}
            queue.put_nowait(self._snapshot())
            return token, queue

    def unsubscribe(self, token):
        with self.condition:
            self.subscribers.pop(token, None)

    def update(self, run, frame=None, image_path=None):
        stream = stream_view(run)
        projected = frame_view(run, frame) if run and frame else None
        with self.condition:
            previous_session = self.stream.get("session_id") if self.stream else None
            self.stream = stream
            if stream is None or stream["session_id"] != previous_session:
                self.frame = None
                self.latest_result_key = None
                self.result_sequence = self.display_sequence = 0
                if self.pending is not None:
                    self.skipped += 1
                    self.pending = None
            if projected is not None:
                self.result_sequence += 1
                self.latest_result_key = projected["frame_key"]
                if projected["status"] == "ok" and image_path is not None:
                    if self.pending is not None:
                        self.skipped += 1
                    self.pending = (projected, self.result_sequence, stream["model"]["version"], image_path)
                    # Keep the latest COMPLETE display frame while newer results
                    # arrive. Faster inference must not cancel every slow encode.
                    if self.frame is None or self.frame["preview_status"] != "ready":
                        self.frame = projected
                    self.condition.notify()
                else:
                    self.frame = projected
                    self.display_sequence = self.result_sequence
                    if self.pending is not None:
                        self.skipped += 1
                        self.pending = None
            self._changed()

    def preview(self, identity):
        with self.condition:
            return self.cache.get(identity)

    def _work(self):
        while True:
            with self.condition:
                while self.pending is None and not self.closed:
                    self.condition.wait()
                if self.closed:
                    return
                projected, sequence, model_version, path = self.pending
                key, session = projected["frame_key"], projected["stream_session_id"]

                if len(data) > self.CACHE_BYTES:
                    raise ValueError("预览超过缓存字节上限")
                identity = uuid.uuid4().hex
                preview = {"id": identity, "url": f"/api/display/previews/{identity}.jpg", "mime": "image/jpeg",
                           "width": size[0], "height": size[1], "sx": size[0]/width, "sy": size[1]/height, "ox": 0, "oy": 0}
                with self.condition:
                    if (not self.stream or self.stream["session_id"] != session
                            or self.stream["model"]["version"] != model_version or sequence <= self.display_sequence):
                        self.skipped += 1
                        continue
                    while self.cache and (len(self.cache) >= self.CACHE_ITEMS or self.cache_bytes+len(data) > self.CACHE_BYTES):
                        _, old = self.cache.popitem(last=False)
                        self.cache_bytes -= len(old)
                    self.cache[identity] = data
                    self.cache_bytes += len(data)
                    self.frame = projected
                    self.display_sequence = sequence
                    self.frame.update(preview=preview, preview_status="ready")
                    self._changed()
            except Exception:
                LOG.exception("预览生成失败（检测及归档结果不受影响）")
                with self.condition:
                    self.errors += 1

                        # A newer successful preview, when available, must remain
                        # intact; otherwise expose the failed frame explicitly.
                        if not self.frame or self.frame["preview_status"] != "ready":
                            self.frame = projected
                            self.frame.update(preview=None, preview_status="error")
                        self._changed()

class UploadSizeLimit:
    """在multipart解析之前限制实际请求字节，包括无Content-Length的分块上传。"""
    def __init__(self, app):
        self.app = app

    async def __call__(self, scope, receive, send):
        if scope["type"] != "http" or scope["path"] != "/api/uploads":
            await self.app(scope, receive, send)
            return
        received = 0

        async def limited_receive():
            nonlocal received
            message = await receive()
            received += len(message.get("body", b""))
            if received > LIMITS["max_file_bytes"] + 1024 * 1024:
                fail(413, "UPLOAD_TOO_LARGE", "上传请求超过21 MiB（图片本身最多20 MiB）")
            return message

为什么需要过渡状态

running 收到 pause 后先进入 pausing，因为当前图片可能还在 PyTorch 原生调用里。前端显示“正在暂停”比立刻显示“已暂停”更准确。类似地，stopping 不等于线程已经结束。

condition.wait 会在等待期间释放对应锁，唤醒后重新取得锁；如果用持锁死循环等待，控制请求就没有机会取得锁并改变状态。

验收与故障定位

暂停后处理数量稳定；继续后增长；停止后历史保留；终态不能 resume，只能新建 run_id。如果任务很短已经结束，pause 可能得到 409，或最后一帧后直接 completed，不要把它认定为暂停逻辑必错。

The checkpoint is a trusted local PyTorch full-model checkpoint, not a user upload.
Call load/predict from the application's single inference worker, never concurrently.

import hashlib
import math
import os
from pathlib import Path
import sys
from time import perf_counter

import numpy as np


VENDOR = Path(__file__).resolve().parents[2] / "vendor" / "yolov5_obb"
# Stable colors are metadata; the UI must not assume a particular class count.
COLORS = ("#087f8c", "#c46a20", "#7561b3", "#397a35", "#bb4d63", "#315fb1", "#8b6c36")


class PytorchOBBAdapter:
    def __init__(self, weights: Path):
        self.weights = Path(weights).resolve()
        self.model = None
        self.metadata = None

    def load(self) -> dict:
        """Load once on CPU and warm up; neither step contributes to per-frame timing."""
        if self.model is not None:
            return deepcopy(self.metadata)
        if self.weights.suffix.lower() != ".pt" or not self.weights.is_file():
            raise ValueError(f"本版需要已存在的本地 .pt 权重：{self.weights}")

        # but refuse to silently reuse a different YOLO repository in this process.
        for name in ("models", "utils"):
            module = sys.modules.get(name)
            if module is not None and getattr(module, "__file__", None):

        from utils.general import non_max_suppression_obb, scale_polys
        from utils.rboxs_utils import rbox2poly

        self.torch = torch
        self.letterbox = letterbox
        self.nms = non_max_suppression_obb
        self.scale_polys = scale_polys
        self.rbox2poly = rbox2poly
        threads = min(4, os.cpu_count() or 1)
        torch.set_num_threads(threads)
        start = perf_counter()
        model = DetectMultiBackend(str(self.weights), device=torch.device("cpu"))
        model.model.float().eval()
        raw_names = model.names
        names = [str(raw_names[i]) for i in range(len(raw_names))]
        if not names or len(names) != int(model.model.model[-1].nc):
            raise ValueError("权重类别表与检测头类别数量不一致")
        # Fail on HBB weights rather than interpreting ordinary output as CSL angles.
        with torch.inference_mode():

            raise ValueError("模型输出不符合 YOLOv5-OBB CSL-180 协议")

        digest = hashlib.sha256()

            for chunk in iter(lambda: stream.read(1024 * 1024), b""):
                digest.update(chunk)
        self.metadata = {
            "name": self.weights.stem,
            "path": str(self.weights),
            "sha256": digest.hexdigest(),
            "classes": [

                for i, name in enumerate(names)
            ],
            "box_type": "obb",
            "device": "cpu",
            "precision": "float32",
            "cpu_threads": threads,
            "stride": int(model.stride),
            "load_warmup_ms": round(load_ms, 3),
        }
        self.model = model
        return deepcopy(self.metadata)

    def predict(self, image_bgr: np.ndarray, *, image_size: int,
                confidence: float, iou: float, max_det: int) -> dict:
        """Return ordered four-corner polygons in original-image pixels.

        Class-aware multi-label NMS, BGR->RGB and auto-stride letterbox match
        detect.py's PyTorch branch. Coordinates intentionally are not clipped.
        """
        if self.model is None:
            raise RuntimeError("请先调用 load() 加载模型")
        if (not isinstance(image_bgr, np.ndarray) or image_bgr.dtype != np.uint8
                or image_bgr.ndim != 3 or image_bgr.shape[2] != 3
                or min(image_bgr.shape[:2]) <= 0):
            raise ValueError("输入必须是非空 uint8 H×W×3 BGR 图片")
        if not 0 <= confidence <= 1 or not 0 <= iou <= 1:
            raise ValueError("confidence 和 iou 必须在 0 到 1 之间")
        if not isinstance(image_size, int) or not 32 <= image_size <= 1280 or image_size % self.metadata["stride"]:
            raise ValueError("image_size 必须在 32..1280 且是模型 stride 的整数倍")
        if not isinstance(max_det, int) or not 1 <= max_det <= 1000:
            raise ValueError("max_det 必须在 1..1000 之间")

        torch = self.torch
        with torch.inference_mode():
            t0 = perf_counter()
            im = self.letterbox(image_bgr, (image_size, image_size),
                                stride=self.metadata["stride"], auto=True)[0]
            im = np.ascontiguousarray(im.transpose((2, 0, 1))[::-1])
            tensor = torch.from_numpy(im).float().div_(255).unsqueeze(0)
            t1 = perf_counter()
            prediction = self.model(tensor, augment=False, visualize=False)
            t2 = perf_counter()
            if not torch.isfinite(prediction).all():
                raise ValueError("模型产生非有限数值，当前图片不记录为成功结果")
            det = self.nms(prediction, confidence, iou, multi_label=True, max_det=max_det)[0]
            polygons = self.rbox2poly(det[:, :5])
            if len(det):
                polygons = self.scale_polys(tensor.shape[2:], polygons, image_bgr.shape)
            detections = []
            for index, (row, polygon) in enumerate(zip(det.tolist(), polygons.tolist())):
                class_id, score = int(row[-1]), float(row[-2])
                if (not 0 <= class_id < len(self.metadata["classes"])
                        or not 0 <= score <= 1 or not all(math.isfinite(x) for x in polygon)):
                    raise ValueError("检测输出含非法类别、置信度或坐标")
                detections.append({
                    "id": index + 1,
                    "class_id": class_id,
                    "class_name": self.metadata["classes"][class_id]["name"],
                    "confidence": score,
                    "polygon": [[polygon[i], polygon[i + 1]] for i in range(0, 8, 2)],
                })
            t3 = perf_counter()
        return {
            "detections": detections,
            "timing_ms": {"preprocess": (t1 - t0) * 1000,
                          "inference": (t2 - t1) * 1000,
                          "postprocess": (t3 - t2) * 1000},
        }

export function decodeImage(url, signal) {
  return new Promise((resolve, reject) => {
    const image = new Image()
    const abort = () => { image.src = ''; reject(new Error('图片加载已取消')) }
    const finish = fn => { signal.removeEventListener('abort', abort); image.onload = image.onerror = null; fn() }
    image.onload = () => finish(() => resolve({ width: image.naturalWidth, height: image.naturalHeight }))
    image.onerror = () => finish(() => reject(new Error('图片解码失败')))
    signal.addEventListener('abort', abort, { once: true })
    if (signal.aborted) abort()
    else image.src = url
  })
}

export function createFrameLoader({ onState, origin, strategy = 'replace', fetchImage = fetch, decode = decodeImage, createUrl = URL.createObjectURL, revokeUrl = URL.revokeObjectURL, timeoutMs = 10000 }) {
  let generation = 0, controller = null, shownUrl = '', shown = null, pendingKey = '', pendingSession = null, queued = null, disposed = false
  function clear() { if (shownUrl) revokeUrl(shownUrl); shownUrl = ''; shown = null }
  function cancelPending() { generation++; controller?.abort(); controller = null; pendingKey = ''; pendingSession = null; queued = null }
  async function load(input) {
    if (disposed) return
    const key = input?.preview ? `${input.key}/${input.preview.id}` : ''
    if (key && key === pendingKey) return
    if (key && key === shown?.loadKey) return
    const sameSession = input && input.sessionId === (pendingSession ?? shown?.sessionId)
    if (strategy === 'latest' && sameSession && input.status === 'ok') {
      if (controller && input.preview) {
        // Do not cancel work at every new frame: that starves a slow decoder.
        // One in-flight image plus the newest waiting input bounds memory.
        queued = input
        onState({ frame: shown, loading: true, error: '' })
        return
      }
      if (!input.preview) {
        onState({ frame: shown, loading: input.previewStatus === 'pending', error: input.previewStatus === 'error' ? '新帧预览生成失败，保留最后完整画面。' : '' })
        return
      }
    }
    cancelPending()
    const keepShown = strategy === 'latest' && sameSession
    if (!input || input.status !== 'ok' || !input.preview) {
      if (!keepShown) clear()
      onState({ frame: shown, loading: input?.status === 'ok' && input.previewStatus === 'pending', error: input?.error || (input?.previewStatus === 'error' ? '预览生成失败；检测结果不等于空检测。' : '') })
      return
    }
    if (!keepShown) clear()
    pendingKey = key
    pendingSession = input.sessionId
    const token = generation, abort = new AbortController()
    controller = abort
    onState({ frame: shown, loading: true, error: '' })
    const timer = setTimeout(() => abort.abort(), timeoutMs)
    let objectUrl = ''
    try {
      const response = await fetchImage(previewUrl(input.preview.url, origin), { signal: abort.signal })
      if (!response.ok) throw new Error(response.status === 404 || response.status === 410 ? '预览已过期或不存在，等待新画面。' : `预览读取失败（HTTP ${response.status}）`)
      objectUrl = createUrl(await response.blob())
      const size = await decode(objectUrl, abort.signal)
      if (size.width !== input.preview.width || size.height !== input.preview.height) throw new Error('预览实际尺寸与协议不一致')
      if (disposed || token !== generation) return
      const p = input.preview
      if (shownUrl) revokeUrl(shownUrl)
      shownUrl = objectUrl; objectUrl = ''
      shown = { ...input, loadKey: key, displayUrl: shownUrl, imageRect: { x: -p.ox / p.sx, y: -p.oy / p.sy, width: p.width / p.sx, height: p.height / p.sy } }
      onState({ frame: shown, loading: false, error: '' })
    } catch (error) {
      if (!disposed && token === generation) onState({ frame: shown, loading: false, error: abort.signal.aborted ? '预览读取超时，等待新画面或重新连接。' : error.message })
    } finally {
      clearTimeout(timer)
      if (objectUrl) revokeUrl(objectUrl)
      if (token === generation) {
        controller = null; pendingKey = ''; pendingSession = null
        const next = queued; queued = null
        if (next) void load(next)
      }
    }
  }
  return { load, disconnect() { cancelPending(); onState({ frame: shown, loading: false, error: '' }) }, dispose() { disposed = true; cancelPending(); clear() } }
}

// 每个请求都有限时；业务错误和断线统一转为可读消息。
export async function api(path, options = {}) {
  const response = await fetch(`/api${path}`, { ...options, signal: AbortSignal.timeout(10000) })
  const contentType = response.headers.get('content-type') || ''
  const data = contentType.includes('json') ? await response.json() : null
  if (!response.ok) {
    const detail = data?.detail
    throw new Error(detail?.message || (typeof detail === 'string' ? detail : `请求失败（HTTP ${response.status}）`))
  }
  if (!data) throw new Error('服务返回了非 JSON 响应，请检查 API 地址。')
  return data
}

export const post = (path, body) => api(path, {
  method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(body),
})

export async function download(url, fallbackName) {
  const response = await fetch(url, { signal: AbortSignal.timeout(60000) })
  if (!response.ok) {
    const data = await response.json().catch(() => null)
    throw new Error(data?.detail?.message || `导出失败（HTTP ${response.status}）`)
  }
  const blobUrl = URL.createObjectURL(await response.blob())
  const link = document.createElement('a')
  link.href = blobUrl
  link.download = fallbackName
  link.click()
  setTimeout(() => URL.revokeObjectURL(blobUrl), 1000)
}
